const int tempPin = A0;
const int soilPin = A1;

void setup()
{
  Serial.begin(9600);
}

void loop(){
  int tempValue = analogRead(tempPin);

  float voltage = tempValue * (5.0 / 1023.0);

  float temp = (voltage - 0.5) * 100.0;

  int soil = analogRead(soilPin);
  int soilP = map(soil, 0,870, 0, 100);

  Serial.print("Temperature: ");
  Serial.print(temp);
  Serial.print(" C");

  Serial.print(" | ");

  Serial.print("Soil: ");
  Serial.print(soilP);
  Serial.println(" %");

  delay(1000);
}
