const int segPins[7] = {0, 1, 2, 3, 4, 5, 6};  // a - g
const int commonPin = 10;                       // common cathode

const int colPins[4] = {7, 8, 9, 11};
const int rowPins[4] = {A0, A1, A2, A3};

const byte seg[16] = {
  0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, // 0 - 7
  0x7F, 0x6F, 0x77, 0x7C, 0x39, 0x5E, 0x79, 0x71  // 8,9,A,b,C,d,E,F
};

const int keymap[4][4] = {
  { 1,  2,  3, 10},
  { 4,  5,  6, 11},
  { 7,  8,  9, 12},
  {14,  0, 15, 13}
};

void show(int v) {
  for (int i = 0; i < 7; i++)
    digitalWrite(segPins[i], (seg[v] >> i) & 1);
}

void setup() {
  pinMode(commonPin, OUTPUT);
  digitalWrite(commonPin, LOW);                 // acts as GND

  for (int i = 0; i < 4; i++) {
    pinMode(colPins[i], OUTPUT);
    digitalWrite(colPins[i], HIGH);
    pinMode(rowPins[i], INPUT_PULLUP);
  }
  for (int i = 0; i < 7; i++) pinMode(segPins[i], OUTPUT);
  show(0);
}

void loop() {                                   // loop กด keypad
  for (int c = 0; c < 4; c++) {
    digitalWrite(colPins[c], LOW);
    for (int r = 0; r < 4; r++) {
      if (digitalRead(rowPins[r]) == LOW) {
        show(keymap[r][c]);
        delay(20);
      }
    }
    digitalWrite(colPins[c], HIGH);
  }
}
