#include <Adafruit_LiquidCrystal.h>

Adafruit_LiquidCrystal lcd(0);   // LCD I2C (address 0)

const int sensorPin = A0;

// custom font กว้าง 5 สูง 8 แถว
byte heart[8] = {
  B00000,
  B01010,
  B11111,
  B11111,
  B01110,
  B00100,
  B00000,
  B00000
};

byte degree[8] = {
  B00110,
  B01001,
  B01001,
  B00110,
  B00000,
  B00000,
  B00000,
  B00000
};

void setup() {
  lcd.begin(16, 2);
  lcd.createChar(0, heart);      // หัวใจตำแหน่ง 0
  lcd.createChar(1, degree);     // องศาตำแหน่ง 1

  // "I l it" (6 ตัว -> เริ่มคอลัมน์ (16-6)/2 = 5)
  lcd.setCursor(5, 0);
  lcd.print("I ");
  lcd.write((uint8_t)0);         // วาดหัวใจ
  lcd.print(" IT");
}

void loop() {
  int raw = analogRead(sensorPin);
  float voltage = raw * 5.0 / 1024.0;
  float tempC = (voltage - 0.5) * 100.0;

  // 181 + temp
  lcd.setCursor(0, 1);
  lcd.print("68070181 ");
  lcd.print(tempC, 2);
  lcd.write((uint8_t)1);
  lcd.print(" ");

  delay(500);
}
