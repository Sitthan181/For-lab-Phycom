#include <stdio.h>
#include <string.h>

int main() {
    int max_size;
    if (scanf("%d", &max_size) != 1) return 0;

    char stack[100][100]; // เก็บข้อความได้สูงสุด 100 คำ แต่ละคำยาวไม่เกิน 100 ตัวอักษร
    int top = 0;          // นับจำนวนข้อมูลปัจจุบันใน Stack
    char input[100];

    while (1) {
        scanf("%s", input);
        
        // ถ้าเจอคำสั่ง 'q' ให้จบการทำงานของลูป
        if (strcmp(input, "q") == 0) {
            break;
        }

        // ถ้าเจอคำสั่ง 'w:' ให้รับชื่อข้อมูลถัดไป
        if (strcmp(input, "w:") == 0) {
            char value[100];
            scanf("%s", value);

            // เช็คกรณี Stack Overflow
            if (top >= max_size) {
                printf("ERROR: Stack Overflow\n");
            } else {
                strcpy(stack[top], value);
                top++;
            }
        }
    }

    // แสดงผลลัพธ์ข้อมูลใน Stack
    for (int i = 0; i < top; i++) {
        printf("%s", stack[i]);
        if (i < top - 1) {
            printf(" -> ");
        }
    }
    printf("\n");

    return 0;
}
